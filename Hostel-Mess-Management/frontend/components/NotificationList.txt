import useSocket from "../hooks/useSocket";

export default function NotificationList() {
  const notifications = useSocket();

  return (
    <div>
      <h2>Notifications</h2>
      <ul>
        {notifications.map((n, i) => (
          <li key={i}>{n}</li>
        ))}
      </ul>
    </div>
  );
}
